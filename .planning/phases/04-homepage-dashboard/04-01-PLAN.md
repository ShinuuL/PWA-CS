    ```sql
      SELECT id, url, caption, created_at
      FROM album_photos
      WHERE pair_id = p_pair_id
      ORDER BY random()
      LIMIT 1;
    $$;
    GRANT EXECUTE ON FUNCTION get_random_album_photo(UUID) TO authenticated;
    ```
